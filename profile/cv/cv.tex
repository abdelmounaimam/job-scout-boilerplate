% The bullets of an entry live in a MINIPAGE: never split across two pages, and

% Two-column entry: text on the left, dates right-aligned.
\newenvironment{twocolentry}[1]{%
  \def\secondColumn{#1}%
  \par
  \setcolumnwidth{\fill,4.5cm}%
  \begin{paracol}{2}%
    \raggedright
}{%
    \switchcolumn
    \raggedleft\strut\secondColumn\par
  \end{paracol}\par\nopagebreak
}

}{
    \par\kern\topsep
